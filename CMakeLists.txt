cmake_minimum_required(VERSION 3.28)
project(LCFCppLibs LANGUAGES CXX)

# Explicit selection overrides registered library submodules. Never scan arbitrary folders.
set(registered_libraries)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/.gitmodules")
    file(STRINGS .gitmodules module_paths REGEX "^[ \t]*path[ \t]*=")
    foreach(line IN LISTS module_paths)
        string(REGEX REPLACE "^[ \t]*path[ \t]*=[ \t]*" "" path "${line}")
        string(STRIP "${path}" path)
        list(APPEND registered_libraries "${path}")
    endforeach()
endif()
if(NOT DEFINED LCF_LIBRARIES)
    set(LCF_LIBRARIES "${registered_libraries}")
endif()
option(LCF_BUILD_EXAMPLES "Enable cross-library experiments" ON)

# Enables discovery when a selected library explicitly enables its own developer tests.
enable_testing()
foreach(path IN LISTS LCF_LIBRARIES)
    if(IS_ABSOLUTE "${path}")
        set(source "${path}")
    else()
        set(source "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
    endif()
    if(NOT EXISTS "${source}/CMakeLists.txt")
        message(FATAL_ERROR "Missing library ${path}; initialize its submodule or fix LCF_LIBRARIES")
    endif()
    get_filename_component(name "${source}" NAME)
    add_subdirectory("${source}" "libraries/${name}")
endforeach()
if(LCF_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
